cmake_minimum_required(VERSION 3.16)
project(pigaze LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

file(GLOB CORE_SOURCES CONFIGURE_DEPENDS src/core/*.cpp)
add_library(pigaze_core STATIC ${CORE_SOURCES})
target_include_directories(pigaze_core PUBLIC include)

enable_testing()
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
add_executable(pigaze_tests ${TEST_SOURCES})
target_link_libraries(pigaze_tests PRIVATE pigaze_core)
add_test(NAME pigaze_tests COMMAND pigaze_tests)

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(Threads REQUIRED)
  file(GLOB LINUX_SOURCES CONFIGURE_DEPENDS src/linux/*.cpp)
  add_executable(pigaze src/main.cpp ${LINUX_SOURCES})
  target_link_libraries(pigaze PRIVATE pigaze_core Threads::Threads)
  install(TARGETS pigaze DESTINATION bin)
endif()
